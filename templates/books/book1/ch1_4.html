Introduction

Arranging everything into state-space form, we get \[ \underbrace{ \frac{d}{dt} \begin{bmatrix} \omega \\ i \end{bmatrix}}_{ \dot{\mathbf{x}}(t)}= \underbrace{ \begin{bmatrix} -\frac{b}{J} &\frac{K_T}{J} \\ -\frac{K_e}{L} & -\frac{R}{L} \end{bmatrix}}_{A} \underbrace{ \begin{bmatrix} \omega \\ i \end{bmatrix}}_{\mathbf{x}(t)}+ \underbrace{ \begin{bmatrix} 0 \\ \frac{1}{L} \end{bmatrix}}_{\mathbf{B}} \mathbf{u}(t)\] For the measurement equation \[ \mathbf{y}(t) = \underbrace{\begin{bmatrix} 1 & 0 \end{bmatrix}}_{\mathbf{C}} \mathbf{x}(t) + \mathbf{v}(t), \] we assume we measure the angular velocity \( x_1 = \omega \), and from this, the measurement matrix follows

Motor Parameters

Symbol Description Value Unit
R Armature resistance 1 ohms
L Armature inductance 0.5 H
\(K_t\) Torque constant 0.1 Nm/A
\(K_e\) Back EMF constant 0.1 V/rad/s
J Rotor inertia 0.01 kg·m²

Results

Matlab Code


clear all;
close all;
clc;

% DC Motor Parameters
Ra = 1;     % Armature resistance (ohms)
La = 0.5;   % Armature inductance (H)
Kt = 0.1;   % Torque constant (Nm/A)
Kb = 0.1;   % Back EMF constant (V/rad/s)
J = 0.01;   % Rotor inertia (kg·m^2)
B = 0.01;   % Friction coefficient (Nm·s)

% State-Space Model Matrices
A = [-Ra/La, -Kb/La;
     Kt/J,   -B/J];
B = [1/La; 0];
C = [0, 1]; % Measure only angular velocity
D = 0;

% Discretize the system for Kalman filter implementation
Ts = 0.01; % Sampling time
sysc = ss(A, B, C, D);
sysd = c2d(sysc, Ts);
[Ad, Bd, Cd, Dd] = ssdata(sysd);

% Noise Covariances
Q = diag([1e-5, 1e-5]); % Process noise covariance

encoder_resolution = 2*pi/2^12;
sigma2_theta = encoder_resolution^2 / 12;
sigma2_omega = 2 * sigma2_theta / Ts^2;
R = sigma2_omega;               % Measurement noise covariance

% Initial States
x_hat = [0; 0];          % Initial state estimate
x_true = [0; 0];         % True initial states
P = eye(2);             % Initial error covariance

% Define Time-Varying Input Signal (Step Inputs)
n_steps = 200; % Simulation steps
time = (0:n_steps-1) * Ts;
u = zeros(1, n_steps);    % Initialize input signal
u(time <= 0.5) = 10;      % Step to 10V for 0.5s
u(time > 0.5 & time <= 1.5) = 20; % Step to 20V for next 1s
u(time > 1.5) = 10;       % Back to 10V for the rest

% fs = 1;
% u = 10 + 10 * sin(2*pi*fs*time);

Qfactor = 1;
Rfactor = 1;

% Monte Carlo Parameters
num_MC = 1000; % Number of Monte Carlo runs

nis_all = zeros(num_MC, n_steps); % Store NIS for all runs
nees_all = zeros(num_MC, n_steps); % Store NEES for all runs

R_sys = Rfactor * R;
Q_sys = Qfactor * Q;

% Kf = (lqr(A',C',Q_sys,R_sys))';   % alternatively, possible to design using "LQR" code

for mc = 1:num_MC
    % Initialize variables for each Monte Carlo run
    x_hat = [0; 0];          % Initial state estimate
    x_true = [0; 0];         % True initial states
    P = eye(2);              % Initial error covariance

    % Initialize storage for results
    x_hat_store = zeros(2, n_steps); % Estimated states
    x_true_store = zeros(2, n_steps); % True states
    y_meas_store = zeros(1, n_steps); % Measured output
    rmse = zeros(2,n_steps);


    for k = 1:n_steps
        % Simulate true system
        w_process = sqrt(Q) * randn(2, 1); % Process noise
        x_true = Ad * x_true + Bd * u(k) + w_process;
        v_meas = sqrt(R) * randn; % Measurement noise
        y_meas = Cd * x_true + v_meas;

        % time-update Step
        x_hat = Ad * x_hat + Bd * u(k);
        P = Ad * P * Ad' + Q_sys;

        % Kalman Filter Update Step
        K = P * Cd' / (Cd * P * Cd' + R_sys); % Kalman gain

        % meas-update step
        x_hat = x_hat + K * (y_meas - Cd * x_hat);
        P = (eye(2) - K * Cd) * P;

         % Store results
        x_hat_store(:, k) = x_hat;
        x_true_store(:, k) = x_true;
        y_meas_store(k) = y_meas;

        estimation_error = x_true - x_hat;
        rmse(:,k) = rmse(:,k) + estimation_error.^2;

        % NIS Calculation
        innovation = y_meas - Cd * x_hat;
        S = Cd * P * Cd' + R_sys; % Innovation covariance
        nis_all(mc, k) = innovation' * (S \ innovation);

        % NEES Calculation
        nees_all(mc, k) = estimation_error' * (P \ estimation_error);
    end

    rmse = sqrt(rmse / num_MC);

    %%
    if mc == num_MC

        % Plot Results
        time = (0:n_steps-1) * Ts;

        % Create a figure
        fh = figure;

        % Set the figure size to 10x8 inches
        width = 15; % width in inches
        height = 12; % height in inches
        set(fh, 'Units', 'inches', 'Position', [1, 1, width, height]);


        subplot(3, 1, 1);
        plot(time, u, 'b', 'LineWidth', 1.5); hold on;
        %plot(time, x_hat_store(2, :), 'r--', 'LineWidth', 1.5);
        %xlabel('Time (s)');
        ylabel('Voltage (V)', 'Interpreter', 'latex');
        ylim([0, 25]);
        %legend('u_k', 'Location', 'best');
        title('Input Voltage');
        grid on;

        subplot(3, 1, 2);
        plot(time, x_true_store(1, :), 'b', 'LineWidth', 1.5); hold on;
        plot(time, x_hat_store(1, :), 'r--', 'LineWidth', 1.5);
        xlabel('Time (s)');
        ylabel('Current i_a (A)', 'Interpreter', 'latex');
        legend('True', 'KF Estimate', 'Location', 'best');
        title('Current Estimation');
        grid on;


        subplot(3, 1, 3);
        plot(time, x_true_store(2, :) * 30 / pi, 'b', 'LineWidth', 1.5); hold on;
        plot(time, x_hat_store(2, :) * 30 / pi, 'r--', 'LineWidth', 1.5);
        %xlabel('Time (s)');
        ylabel('Angular Velocity ($\omega$) (rpm)', 'Interpreter', 'latex');
        legend('True', 'KF Estimate', 'Location', 'best');
        title('Angular Velocity');
        grid on;

    end
end

%%
% Compute Mean NIS and NEES
mean_nis = mean(nis_all, 1); % Average NIS at each time step
mean_nees = mean(nees_all, 1); % Average NEES at each time step

% Theoretical Bounds (Chi-Square)
alpha = 0.05; % Confidence level
nis_bounds = chi2inv([alpha/2, 1-alpha/2], num_MC * 1) / num_MC; % Degrees of freedom = 1 (for scalar measurement)
nees_bounds = chi2inv([alpha/2, 1-alpha/2], num_MC * 2) / num_MC; % Degrees of freedom = 2 (for 2 state variables)

% Plot NIS and NEES
time = (0:n_steps-1) * Ts;

fh = figure;
set(fh, 'Units', 'inches', 'Position', [1, 1, width, height]);
subplot(2, 1, 1);
plot(time, mean_nis, 'b', 'LineWidth', 1.5); hold on;
yline(nis_bounds(1), 'r--', 'LineWidth', 1.5);
yline(nis_bounds(2), 'r--', 'LineWidth', 1.5);
xlabel('Time (s)');
ylabel('Mean NIS');
title('Normalized Innovation Squared (NIS)');
legend('Mean NIS', 'Bounds', 'Location', 'best');
grid on;
ylim([0.5, 1.5])

subplot(2, 1, 2);
plot(time, mean_nees, 'b', 'LineWidth', 1.5); hold on;
yline(nees_bounds(1), 'r--', 'LineWidth', 1.5);
yline(nees_bounds(2), 'r--', 'LineWidth', 1.5);
xlabel('Time (s)');
ylabel('Mean NEES');
title('Normalized Estimation Error Squared (NEES)');
legend('Mean NEES', 'Bounds', 'Location', 'best');
grid on;
ylim([1.5, 2.5])
